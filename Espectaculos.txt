CREATE DATABASE ventas_entradas;
USE ventas_entradas;

CREATE TABLE RECINTOS (
    COD_RECINTO, VARCHAR(10) PRIMARY KEY,
    NOMBRE VARCHAR(50) NOT NULL, 
    DIRECCION VARCHAR(255) NOT NULL, 
    CIUDAD VARCHAR(50) NOT NULL, 
    TELEFONO VARCHAR(10)NOT NULL, 
    HORARIO VARCHAR(50) NOT NULL,
);

CREATE TABLE ESPECTADORES (
    DNI_CLIENTE VARCHAR(10) PRIMARY KEY, 
    NOMBRE VARCHAR(50) NOT NULL, 
    DIRECCION VARCHAR(255) NOT NULL, 
    TELEFONO VARCHAR(15) NOT NULL, 
    CIUDAD VARCHAR(50) NOT NULL,
    TARJETA VARCHAR(20) NOT NULL,
    )

CREATE TABLE  ESPECTACULOS (
    COD_ESPECTACULO VARCHAR(10) PRIMARY KEY,
    NOMBRE VARCHAR(50) NOT NULL,
    TIPO VARCHAR(50) NOT NULL,
    FECHA_INICIAL DATE NOT NULL,
    FECHA_FINAL DATE NOT NULL
    INTERPRETE VARCHAR(50) NOT NULL,
    COD_RECINTO VARCHAR(10) NOT NULL,
    FOREIGN KEY (COD_RECINTO) REFERENCES RECINTOS (COD_RECINTO)
);

CREATE TABLE ZONAS_RECINTOS (
    COD_RECINTO VARCHAR(10), 
    ZONA VARCHAR(50), 
    CAPACIDAD INT NOT NULL,
    COD_PUB
    PRIMARY KEY (COD_RECINTO, ZONA),
    FOREIGN KEY (COD_RECINTO) REFERENCES RECINTOS(COD_RECINTO)
);

CREATE TABLE ASIENTOS (
    COD_RECINTO VARCHAR(10), 
    ZONA VARCHAR(50), 
    FILA VARCHAR(10), 
    NUMERO INT,
    PRIMARY KEY (COD_RECINTO, ZONA, FILA),
    FOREIGN KEY (COD_RECINTO, ZONA) REFERENCES ZONAS_RECINTOS (COD_RECINTO, ZONA)
);

CREATE TABLE PRECIOS_ESPECTACULOS (
    COD_ESPECTACULO VARCHAR(10),
    COD_RECINTO INT VARCHAR(10),
    ZONA VARCHAR(50),
    PRECIO DECIMAL(6, 2) NOT NULL,
    PRIMARY KEY (COD_ESPECTACULO, COD_RECINTO, ZONA),
    FOREIGN KEY (COD_ESPECTACULO) REFERENCES ESPECTACULOS(COD_ESPECTACULO),
    FOREIGN KEY (COD_RECINTO, ZONA) REFERENCES ZONAS_RECINTOS (COD_RECINTO, ZONA)
);

CREATE TABLE REPRESENTACIONES (
    COD_ESPECTACULO VARCHAR(10), 
    FECHA DATE,
    HORA VARCHAR(5),
    PRIMARY KEY (COD_ESPECTACULO, FECHA, HORA)
    FOREIGN KEY (COD_ESPECTACULO) REFERENCE ESPECTACULOS (COD_ESPECTACULO),
);

CREATE TABLE ENTRADAS (
    COD_ESPECTACULO VARCHAR(10),
    FECHA           DATE,
    HORA            VARCHAR(5),
    COD_RECINTO     VARCHAR(10),
    ZONA            VARCHAR(30),
    FILA            VARCHAR(5),
    NUMERO          INT,
    DNI_CLIENTE     VARCHAR(10) NOT NULL,
    PRIMARY KEY (COD_ESPECTACULO, FECHA, HORA, COD_RECINTO, ZONA, FILA, NUMERO),
    FOREIGN KEY (COD_ESPECTACULO, FECHA, HORA) REFERENCES REPRESENTACIONES(COD_ESPECTACULO, FECHA, HORA),
    FOREIGN KEY (COD_RECINTO, ZONA, FILA, NUMERO) REFERENCES ASIENTOS(COD_RECINTO, ZONA, FILA, NUMERO),
    FOREIGN KEY (DNI_CLIENTE) REFERENCES ESPECTADORES(DNI_CLIENTE)
);
