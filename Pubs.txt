CREATE TABLE PUB (
COD_PUB VARCHER(10) PRIMARY KEY, NOT NULL, 
Nombre VARCHAR(20) NOT NULL,
Licencia_Fiscal VARCHAR(20) NOT NULL,
Domicilio VARCHAR(255) DEFAULT NULL;
Fecha_Apertura INTERGER NOT NULL;
Horario CHECK (horario in (“Hor1”, “Hor2”, “Hor3”));
COD_Localidad INTERGER NOT NULL,
)


CREATE TITULAR (
DNI_Titular INTERGER(10) PRIMARY KEY, NOT NULL, 
Nombre VARCHAR(20) NOT NULL;
Domicilio VARCHAR(255) DEFAULL NULL;
COD_PUB VARCHER(10) NOT NULL;
)

CREATE EMPLEADO (
Nombre VARCHER(10) PRIMARY KEY, NOT NULL, 
Domicilio VARCHAR(255) DEFAULT NULL; 
Pub_empleado CHECK (puesto in (“CAMARERO” “SEGURIDAD” “LIMPIEZA”));
)

CREATE EXISTENCIAS (
Codigo_Articulo VARCHER(10) PRIMARY KEY, NOT NULL,
Nombre VARCHER(20);
Cantidad INTERGER();
Precio DECIMAL (10, 2) NOT NULL;
COD_PUB VARCHER(10);
)

CREATE LOCALIDAD (
Nombre
COD_LOCALIDAD PRIMARY KEY,
)

CREATE PUB_EMPLEADO (
Funcion PRIMARY KEY,
COD PRIMARY KEY,
DNI_Empleado PRIMARY KEY,
)
